def seed_demo_users() -> None:
    # Zero-friction local dev ke liye tables ensure kar lo (agar app kabhi
    # run nahi hua to bhi ye script akele chal jaye).
    Base.metadata.create_all(bind=engine)

    db = SessionLocal()
    added, skipped = 0, 0

    try:
        for u in DEMO_USERS:
            existing = db.query(User).filter(User.email == u["email"]).first()
            if existing:
                skipped += 1
                continue

            user = User(
                full_name=u["full_name"],
                email=u["email"],
                hashed_password=hash_password(DEMO_PASSWORD),
                role=u["role"],
            )
            db.add(user)
            db.flush()  # user.id chahiye moderator permissions ke liye

            if u["role"] == UserRole.MODERATOR:
                perms = u.get("permissions", {})
                db.add(
                    ModeratorPermission(
                        user_id=user.id,
                        can_manage_users=perms.get("can_manage_users", False),
                        can_manage_courses=perms.get("can_manage_courses", False),
                        can_manage_resources=perms.get("can_manage_resources", False),
                        can_manage_enrollments=perms.get("can_manage_enrollments", False),
                        can_view_reports=perms.get("can_view_reports", False),
                    )
                )

            added += 1

        db.commit()
    finally:
        db.close()

    print(f"✅ {added} naye demo users add hue.")
    if skipped:
        print(f"⏭️  {skipped} users pehle se maujood the (email match), unhe skip kar diya.")
    print(f"🔑 Sabka demo password: {DEMO_PASSWORD}")


def show_users() -> None:
    db = SessionLocal()
    try:
        users = db.query(User).order_by(User.id).all()
        print(f"\n📋 Total users in database: {len(users)}\n")
        print(f"{'ID':<4} {'Name':<20} {'Email':<32} {'Role':<12} Active")
        print("-" * 85)
        for u in users:
            print(f"{u.id:<4} {u.full_name:<20} {u.email:<32} {u.role.value:<12} {u.is_active}")
    finally:
        db.close()


if __name__ == "__main__":
    seed_demo_users()
    show_users()
    print("\n🎉 Done! Demo users successfully seeded.")
